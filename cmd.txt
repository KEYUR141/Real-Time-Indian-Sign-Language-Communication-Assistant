nvidia-smi

# 5. TRAIN ALL THREE MAIN MODELS IN ONE SEQUENTIAL BATCH
# Run section 0 first. This launches SVM, then LSTM, then Transformer.
# This is sequential, not three simultaneous GPU jobs and not an ensemble.
# The API demonstration uses an existing checkpoint; retraining is NOT needed
# before presenting. Stop GPU-heavy work before timing live inference.
# Each run gets a new output folder to protect the recorded benchmark artifacts.
$RunRoot = "$FeatureRoot\models\presentation_runs\$(Get-Date -Format 'yyyyMMdd_HHmmss_fff')"
$EvaluateTest = $false
# Default: validation only. For an explicitly planned final test run, change the
# value above to $true. Existing published test results are already frozen.
# --evaluate-test retrains/refits and evaluates; it does NOT merely load a .pt.
$TestArgs = @()
if ($EvaluateTest) { $TestArgs = @("--evaluate-test") }

uv run python scripts/train_svm_baseline.py --config "$Config" --output "$RunRoot\svm" @TestArgs
if ($LASTEXITCODE -ne 0) { throw "SVM failed; batch stopped." }

uv run python scripts/train_temporal_baselines.py --config "$Config" --models lstm --epochs 140 --patience 20 --seed 42 --device cuda --output "$RunRoot\lstm_seed_42" @TestArgs
if ($LASTEXITCODE -ne 0) { throw "LSTM failed; batch stopped." }

uv run python scripts/train_temporal_baselines.py --config "$Config" --models transformer --epochs 80 --patience 12 --seed 42 --device cuda --output "$RunRoot\transformer_seed_42" @TestArgs
if ($LASTEXITCODE -ne 0) { throw "Transformer failed; batch stopped." }

Write-Output "All three completed. Reports: $RunRoot"
# LSTM uses the documented 140-epoch cap and patience 20. Transformer uses
# hidden size 128, 4 heads, 2 layers, feed-forward size 256, dropout 0.3.
# A new run may not reproduce every decimal, even with seed 42 on CUDA.

# 6. TRAIN EACH MODEL SEPARATELY (choose only the desired command)
# Run section 0 first. Use a fresh experiment folder for each new run.
$SeparateRoot = "$FeatureRoot\models\separate_runs\$(Get-Date -Format 'yyyyMMdd_HHmmss_fff')"

# SVM: Support Vector Machine; CPU-based scikit-learn trainer.
uv run python scripts/train_svm_baseline.py --config "$Config" --output "$SeparateRoot\svm"

# LSTM: Long Short-Term Memory; sequence model.
uv run python scripts/train_temporal_baselines.py --config "$Config" --models lstm --epochs 140 --patience 20 --seed 42 --device cuda --output "$SeparateRoot\lstm_seed_42"

# Compact Transformer: temporal attention model.
uv run python scripts/train_temporal_baselines.py --config "$Config" --models transformer --epochs 80 --patience 12 --seed 42 --device cuda --output "$SeparateRoot\transformer_seed_42"

# GRU: Gated Recurrent Unit; optional additional temporal comparison.
uv run python scripts/train_temporal_baselines.py --config "$Config" --models gru --epochs 80 --patience 12 --seed 42 --device cuda --output "$SeparateRoot\gru_seed_42"

# To intentionally refit and test any one model, append --evaluate-test to its
# command. Never use test scores to repeatedly choose configurations or seeds.
# SVM does not accept --device or --seed. Neural trainers accept both.

# 7. COMPARE GRU + LSTM + TRANSFORMER IN ONE TRAINER INVOCATION
# Run section 0 first. SVM has its own trainer and cannot go in --models.
uv run python scripts/train_temporal_baselines.py --config "$Config" --models gru lstm transformer --epochs 140 --patience 20 --seed 42 --device cuda --output "$FeatureRoot\models\temporal_comparison_$(Get-Date -Format 'yyyyMMdd_HHmmss_fff')"
# Models train one after another and receive the SAME supplied settings here.
# Output: one combined report and a validation checkpoint for each architecture.
# Appending --evaluate-test tests ONLY the validation-selected architecture.
# Use section 5 if you need independent final test reports for all three models.

# 8. INCLUDE-50 EXPERIMENTS (choose this configuration intentionally)
$Config = "configs/landmarks.toml"

# INCLUDE-50 and full INCLUDE share the configured feature root but use their
# own manifest-selected sample identities, split metadata, and label maps.

# 9. INSPECT EXISTING RESULTS WITHOUT TRAINING

# Final neural checkpoint after --evaluate-test: selected_final.pt
